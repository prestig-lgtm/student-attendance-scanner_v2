/** * ============================================================ * PSABE-PPG CSU CHAPTER ATTENDANCE SYSTEM * GOOGLE APPS SCRIPT BACKEND * ============================================================ * * MASTERLIST / YEAR-LEVEL SHEETS: * First Years * Second Years * Third Years * Fourth Years * * Existing year-level structure is preserved: * A = STUDENT * B = ID NUMBER * C = TIME IN * D = TIME OUT * E = LATE/NO TIME IN * F = ATTENDANCE STATUS * * New event attendance sheets are created automatically. * Event sheet structure: * Row 1 = Event / Activity Name * Row 2 = Student ID | Name | Time In | Time Out * Row 3 onward = Attendance records * * Supported POST actions: * LOOKUP -> find student name from the year-level masterlists * CREATE_SHEET -> create a new event attendance sheet * TIME IN -> record/update Time In * TIME OUT -> record/update Time Out * * Supported GET actions: * lookup * listSheets * getRecords */ const YEAR_SHEETS = [ "First Years", "Second Years", "Third Years", "Fourth Years" ]; const STUDENT_COLUMN = 1; const ID_COLUMN = 2; const TIME_IN_COLUMN = 3; const TIME_OUT_COLUMN = 4; const REMARK_COLUMN = 5; const STATUS_COLUMN = 6; const HEADER_ROW = 2; const FIRST_STUDENT_ROW = 3; const EVENT_HEADER_ROW = 2; const EVENT_FIRST_DATA_ROW = 3; const EVENT_HEADERS = ["Student ID", "Name", "Time In", "Time Out"]; const MIN_TIME_OUT_MINUTES = 60; function getSpreadsheet() { return SpreadsheetApp.getActiveSpreadsheet(); } function jsonResponse(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } function cleanValue(value) { if (value === null || value === undefined) return ""; return String(value).trim(); } function getCurrentTime() { const timezone = getSpreadsheet().getSpreadsheetTimeZone(); return Utilities.formatDate(new Date(), timezone, "h:mm:ss a"); } function getToday() { const timezone = getSpreadsheet().getSpreadsheetTimeZone(); return Utilities.formatDate(new Date(), timezone, "yyyy-MM-dd"); } /* ============================================================ MASTERLIST LOOKUP ============================================================ */ function findStudent(studentId) { const ss = getSpreadsheet(); const cleanId = cleanValue(studentId).toUpperCase(); if (!cleanId) return null; for (const sheetName of YEAR_SHEETS) { const sheet = ss.getSheetByName(sheetName); if (!sheet) continue; const lastRow = sheet.getLastRow(); if (lastRow < FIRST_STUDENT_ROW) continue; const header = cleanValue( sheet.getRange(HEADER_ROW, ID_COLUMN).getValue() ).toUpperCase(); if (header !== "ID NUMBER") continue; const numberOfStudents = lastRow - FIRST_STUDENT_ROW + 1; const values = sheet.getRange( FIRST_STUDENT_ROW, 1, numberOfStudents, 2 ).getDisplayValues(); for (let i = 0; i < values.length; i++) { const existingId = cleanValue(values[i][1]).toUpperCase(); if (existingId !== cleanId) continue; const row = FIRST_STUDENT_ROW + i; return { sheet, row, studentName: cleanValue(values[i][0]), studentId: cleanValue(values[i][1]), yearLevel: sheetName }; } } return null; } /* ============================================================ TIME HELPERS ============================================================ */ function parseTimeToMinutes(value) { const text = cleanValue(value).toUpperCase(); if (!text) return null; // h:mm[:ss] AM/PM let match = text.match(/^(\d{1,2}):(\d{2})(?::\d{2})?\s*(AM|PM)$/); if (match) { let hour = Number(match[1]); const minute = Number(match[2]); const meridiem = match[3]; if (hour === 12) hour = 0; if (meridiem === "PM") hour += 12; return hour * 60 + minute; } // 24-hour HH:mm[:ss] match = text.match(/^(\d{1,2}):(\d{2})(?::\d{2})?$/); if (match) { return Number(match[1]) * 60 + Number(match[2]); } return null; } function isTimeOutAllowed(timeIn, timeOut) { const inMinutes = parseTimeToMinutes(timeIn); const outMinutes = parseTimeToMinutes(timeOut); if (inMinutes === null || outMinutes === null) return true; // Handles a midnight crossing if ever needed. let difference = outMinutes - inMinutes; if (difference < 0) difference += 24 * 60; return difference >= MIN_TIME_OUT_MINUTES; } function calculateStatus(timeIn, timeOut) { if (timeIn && timeOut) return "COMPLETED"; if (timeOut && !timeIn) return "INCOMPLETE"; if (timeIn && !timeOut) return "PRESENT"; return ""; } /* ============================================================ EVENT SHEET HELPERS ============================================================ */ function sanitizeSheetName(name) { let result = cleanValue(name) .replace(/[:\\/?*\[\]]/g, "-") .replace(/\s+/g, " ") .trim(); if (!result) result = "Attendance"; if (result.length > 90) result = result.substring(0, 90).trim(); return result; } function makeUniqueSheetName(baseName) { const ss = getSpreadsheet(); const base = sanitizeSheetName(baseName); if (!ss.getSheetByName(base)) return base; let counter = 2; while (counter < 1000) { const suffix = " (" + counter + ")"; const maxBaseLength = 100 - suffix.length; const candidate = base.substring(0, maxBaseLength).trim() + suffix; if (!ss.getSheetByName(candidate)) return candidate; counter++; } return base.substring(0, 80) + " (" + Date.now() + ")"; } function setupEventSheet(sheet, eventName) { sheet.getRange(1, 1, 1, 4).merge(); sheet.getRange(1, 1).setValue(eventName); sheet.getRange(1, 1).setFontWeight("bold").setFontSize(14); sheet.getRange(1, 1).setHorizontalAlignment("center"); sheet.getRange(1, 1).setNote(new Date().toISOString()); sheet.getRange(EVENT_HEADER_ROW, 1, 1, EVENT_HEADERS.length) .setValues([EVENT_HEADERS]) .setFontWeight("bold") .setBackground("#eaf6ee"); sheet.setFrozenRows(EVENT_HEADER_ROW); sheet.autoResizeColumns(1, EVENT_HEADERS.length); sheet.setColumnWidth(1, Math.max(sheet.getColumnWidth(1), 120)); sheet.setColumnWidth(2, Math.max(sheet.getColumnWidth(2), 220)); sheet.setColumnWidth(3, Math.max(sheet.getColumnWidth(3), 110)); sheet.setColumnWidth(4, Math.max(sheet.getColumnWidth(4), 110)); } function createAttendanceSheet(eventName) { const ss = getSpreadsheet(); const actualName = makeUniqueSheetName(eventName); const sheet = ss.insertSheet(actualName); setupEventSheet(sheet, actualName); return { success: true, sheetId: String(sheet.getSheetId()), sheetName: actualName, createdAt: new Date().toISOString() }; } function isEventAttendanceSheet(sheet) { if (!sheet) return false; if (YEAR_SHEETS.indexOf(sheet.getName()) !== -1) return false; if (sheet.getLastRow() < EVENT_HEADER_ROW) return false; const headers = sheet .getRange(EVENT_HEADER_ROW, 1, 1, 4) .getDisplayValues()[0] .map(v => cleanValue(v).toUpperCase()); return ( headers[0] === "STUDENT ID" && headers[1] === "NAME" && headers[2] === "TIME IN" && headers[3] === "TIME OUT" ); } function findEventRow(sheet, studentId) { if (sheet.getLastRow() < EVENT_FIRST_DATA_ROW) return 0; const count = sheet.getLastRow() - EVENT_FIRST_DATA_ROW + 1; const ids = sheet .getRange(EVENT_FIRST_DATA_ROW, 1, count, 1) .getDisplayValues(); const cleanId = cleanValue(studentId).toUpperCase(); for (let i = 0; i < ids.length; i++) { if (cleanValue(ids[i][0]).toUpperCase() === cleanId) { return EVENT_FIRST_DATA_ROW + i; } } return 0; } function getOrCreateEventSheet(eventName) { const ss = getSpreadsheet(); const cleanName = sanitizeSheetName(eventName); let sheet = ss.getSheetByName(cleanName); if (sheet && isEventAttendanceSheet(sheet)) { return sheet; } if (sheet && !isEventAttendanceSheet(sheet)) { const actualName = makeUniqueSheetName(cleanName); sheet = ss.insertSheet(actualName); setupEventSheet(sheet, actualName); return sheet; } sheet = ss.insertSheet(cleanName); setupEventSheet(sheet, cleanName); return sheet; } function resolveEventSheet(data) { const ss = getSpreadsheet(); const sheetId = cleanValue(data.sheetId || data.googleSheetId); const eventName = cleanValue( data.eventName || data.sheetName || data.attendanceSheetName ); if (sheetId && /^\d+$/.test(sheetId)) { const byId = ss.getSheets().find(s => String(s.getSheetId()) === sheetId); if (byId && isEventAttendanceSheet(byId)) return byId; } if (eventName) { const named = ss.getSheetByName(sanitizeSheetName(eventName)); if (named && isEventAttendanceSheet(named)) return named; return getOrCreateEventSheet(eventName); } return null; } /* ============================================================ YEAR-LEVEL SHEET UPDATE — EXISTING FUNCTIONALITY PRESERVED ============================================================ */ function updateYearLevelAttendance(student, action, timeIn, timeOut, remark, status) { const sheet = student.sheet; const row = student.row; if (action === "TIME IN") { sheet.getRange(row, TIME_IN_COLUMN).setValue(timeIn || ""); if (timeOut) sheet.getRange(row, TIME_OUT_COLUMN).setValue(timeOut); sheet.getRange(row, REMARK_COLUMN).setValue(remark || ""); sheet.getRange(row, STATUS_COLUMN).setValue( status || calculateStatus(timeIn, timeOut) ); } if (action === "TIME OUT") { if (timeIn) sheet.getRange(row, TIME_IN_COLUMN).setValue(timeIn); sheet.getRange(row, TIME_OUT_COLUMN).setValue(timeOut || ""); sheet.getRange(row, REMARK_COLUMN).setValue( remark || (timeIn ? "" : "NO TIME IN") ); sheet.getRange(row, STATUS_COLUMN).setValue( status || calculateStatus(timeIn, timeOut) ); } } /* ============================================================ EVENT SHEET ATTENDANCE UPDATE ============================================================ */ function updateEventAttendance(eventSheet, student, action, timeIn, timeOut) { const row = findEventRow(eventSheet, student.studentId); let targetRow = row; if (!targetRow) { targetRow = Math.max(eventSheet.getLastRow() + 1, EVENT_FIRST_DATA_ROW); eventSheet.getRange(targetRow, 1, 1, 4).setValues([[ student.studentId, student.studentName, timeIn || "", timeOut || "" ]]); } else { // Always refresh the authoritative masterlist name. eventSheet.getRange(targetRow, 1).setValue(student.studentId); eventSheet.getRange(targetRow, 2).setValue(student.studentName); if (timeIn) eventSheet.getRange(targetRow, 3).setValue(timeIn); if (timeOut) eventSheet.getRange(targetRow, 4).setValue(timeOut); } // Explicitly handle blank Time In / Time Out without overwriting // a previously stored value during offline synchronization. if (action === "TIME IN" && timeIn) { eventSheet.getRange(targetRow, 3).setValue(timeIn); } if (action === "TIME OUT" && timeOut) { eventSheet.getRange(targetRow, 4).setValue(timeOut); } return targetRow; } /* ============================================================ LOOKUP / LIST / RECORD GETTERS ============================================================ */ function lookupStudent(studentId) { const student = findStudent(studentId); if (!student) { return { success: false, message: "Student ID " + cleanValue(studentId) + " was not found in the year-level sheets." }; } return { success: true, studentId: student.studentId, studentName: student.studentName, yearLevel: student.yearLevel }; } function listAttendanceSheets() { const sheets = getSpreadsheet().getSheets() .filter(isEventAttendanceSheet) .map(sheet => ({ sheetId: String(sheet.getSheetId()), sheetName: sheet.getName(), createdAt: sheet.getRange(1, 1).getNote() || "" })); return { success: true, sheets }; } function getAttendanceRecords(sheetId, sheetName) { const ss = getSpreadsheet(); let sheet = null; if (cleanValue(sheetId) && /^\d+$/.test(cleanValue(sheetId))) { sheet = ss.getSheets().find(s => String(s.getSheetId()) === cleanValue(sheetId)); } if (!sheet && cleanValue(sheetName)) { sheet = ss.getSheetByName(sanitizeSheetName(sheetName)); } if (!sheet || !isEventAttendanceSheet(sheet)) { return { success: false, message: "Attendance sheet not found.", records: [] }; } if (sheet.getLastRow() < EVENT_FIRST_DATA_ROW) { return { success: true, sheetId: String(sheet.getSheetId()), sheetName: sheet.getName(), records: [] }; } const count = sheet.getLastRow() - EVENT_FIRST_DATA_ROW + 1; const values = sheet.getRange(EVENT_FIRST_DATA_ROW, 1, count, 4).getDisplayValues(); const records = values .filter(row => row.some(v => cleanValue(v) !== "")) .map(row => ({ studentId: cleanValue(row[0]), studentName: cleanValue(row[1]), timeIn: cleanValue(row[2]), timeOut: cleanValue(row[3]), status: calculateStatus(cleanValue(row[2]), cleanValue(row[3])) })); return { success: true, sheetId: String(sheet.getSheetId()), sheetName: sheet.getName(), records }; } /* ============================================================ GET ============================================================ */ function doGet(e) { try { const params = (e && e.parameter) || {}; const action = cleanValue(params.action).toLowerCase(); if (action === "lookup") { return jsonResponse(lookupStudent(params.studentId || params.id)); } if (action === "listsheets") { return jsonResponse(listAttendanceSheets()); } if (action === "getrecords") { return jsonResponse( getAttendanceRecords(params.sheetId, params.sheetName) ); } return jsonResponse({ success: true, message: "Student Attendance System is online." }); } catch (error) { return jsonResponse({ success: false, message: "Apps Script error: " + error.message }); } } /* ============================================================ POST ============================================================ */ function doPost(e) { const lock = LockService.getScriptLock(); try { if (!e || !e.postData || !e.postData.contents) { return jsonResponse({ success: false, message: "No POST body was received." }); } const data = JSON.parse(e.postData.contents); const action = cleanValue(data.action || data.mode).toUpperCase(); /* -------------------------------------------------------- LOOKUP -------------------------------------------------------- */ if (action === "LOOKUP") { return jsonResponse(lookupStudent( data.studentId || data.studentID || data.id || data.ID )); } /* -------------------------------------------------------- CREATE EVENT SHEET -------------------------------------------------------- */ if ( action === "CREATE_SHEET" || action === "CREATE SHEET" || action === "CREATE_ATTENDANCE_SHEET" ) { const eventName = cleanValue( data.eventName || data.sheetName || data.name ); if (!eventName) { return jsonResponse({ success: false, message: "An event/activity name is required." }); } lock.waitLock(10000); try { const result = createAttendanceSheet(eventName); // Store the creation timestamp as a note without changing the visible format. const createdSheet = getSpreadsheet().getSheets() .find(s => String(s.getSheetId()) === result.sheetId); if (createdSheet) { createdSheet.getRange(1, 1).setNote(result.createdAt); } SpreadsheetApp.flush(); return jsonResponse(result); } finally { lock.releaseLock(); } } /* -------------------------------------------------------- ATTENDANCE -------------------------------------------------------- */ if (action !== "TIME IN" && action !== "TIME OUT") { return jsonResponse({ success: false, message: "Invalid attendance action.", receivedAction: action }); } const studentId = cleanValue( data.studentId || data.studentID || data.id || data.ID ); const incomingTimeIn = cleanValue(data.timeIn); const incomingTimeOut = cleanValue(data.timeOut); const incomingRemark = cleanValue(data.remark); const incomingStatus = cleanValue(data.status).toUpperCase(); const eventName = cleanValue( data.eventName || data.sheetName || data.attendanceSheetName ); if (!studentId) { return jsonResponse({ success: false, message: "No Student ID was received." }); } const student = findStudent(studentId); if (!student) { return jsonResponse({ success: false, message: "Student ID " + studentId + " was not found in the year-level sheets.", studentId }); } /* The 1-hour rule is enforced on the server as well as in HTML. */ if (action === "TIME OUT" && incomingTimeIn && incomingTimeOut) { if (!isTimeOutAllowed(incomingTimeIn, incomingTimeOut)) { return jsonResponse({ success: false, message: "Time Out is not allowed until at least 1 hour after Time In.", studentId, studentName: student.studentName, timeIn: incomingTimeIn, timeOut: incomingTimeOut }); } } lock.waitLock(10000); try { const eventSheet = resolveEventSheet(data); let eventRow = null; if (eventSheet) { eventRow = updateEventAttendance( eventSheet, student, action, incomingTimeIn, incomingTimeOut ); } // Preserve the original year-level attendance tracking too. updateYearLevelAttendance( student, action, incomingTimeIn, incomingTimeOut, incomingRemark, incomingStatus || calculateStatus(incomingTimeIn, incomingTimeOut) ); SpreadsheetApp.flush(); const finalStatus = calculateStatus(incomingTimeIn, incomingTimeOut); return jsonResponse({ success: true, message: action + " synchronized for " + student.studentName, studentName: student.studentName, studentId: student.studentId, yearLevel: student.yearLevel, timeIn: incomingTimeIn, timeOut: incomingTimeOut, remark: incomingRemark || (incomingTimeOut && !incomingTimeIn ? "NO TIME IN" : ""), status: incomingStatus || finalStatus, eventSheet: eventSheet ? eventSheet.getName() : "", eventSheetId: eventSheet ? String(eventSheet.getSheetId()) : "", eventRow: eventRow || "", sheet: student.yearLevel, row: student.row }); } finally { lock.releaseLock(); } } catch (error) { console.error(error && error.stack ? error.stack : error); try { lock.releaseLock(); } catch (_) {} return jsonResponse({ success: false, message: "Apps Script error: " + error.message }); } }
